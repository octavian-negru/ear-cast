cmake_minimum_required(VERSION 3.22)
project(earcast_audio_quality LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(EARCAST_BUILD_JNI OFF CACHE BOOL "No JNI needed by the offline suite" FORCE)
add_subdirectory(../audio-engine/src/main/cpp production)
enable_testing()
add_executable(rnnoise_filter_test native/rnnoise_filter_test.cpp)
target_link_libraries(rnnoise_filter_test PRIVATE neural_speech)
add_test(NAME neural_speech COMMAND rnnoise_filter_test)
add_executable(voice_leveler_test native/voice_leveler_test.cpp)
target_link_libraries(voice_leveler_test PRIVATE neural_speech)
add_test(NAME voice_leveler COMMAND voice_leveler_test)
add_executable(classic_filters_test native/classic_filters_test.cpp)
target_link_libraries(classic_filters_test PRIVATE wiener_speech)
add_test(NAME classical_speech COMMAND classic_filters_test)
add_executable(render_audio native/render_audio.cpp)
target_link_libraries(render_audio PRIVATE neural_speech speex_preprocess wiener_speech)

# Optional complete DPDFNet replay uses pre-existing Linux v1.13.8 libraries.
# Nothing is downloaded or built by the evaluation scripts.
if(TARGET detailed_speech)
    target_link_libraries(render_audio PRIVATE detailed_speech)
    target_compile_definitions(render_audio PRIVATE EARCAST_HAS_DPDFNET=1)
    add_executable(dpdfnet_filter_test native/dpdfnet_filter_test.cpp)
    target_link_libraries(dpdfnet_filter_test PRIVATE detailed_speech)
    add_test(NAME dpdfnet_speech COMMAND dpdfnet_filter_test
        ${CMAKE_CURRENT_SOURCE_DIR}/../audio-engine/src/main/assets/speech-models)
endif()
