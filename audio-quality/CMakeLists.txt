cmake_minimum_required(VERSION 3.22)
project(openhearing_audio_quality LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENHEARING_BUILD_JNI OFF CACHE BOOL "No JNI needed by the offline suite" FORCE)
add_subdirectory(../core-audio/src/main/cpp production)
enable_testing()
add_executable(neural_speech_test native/neural_speech_test.cpp)
target_link_libraries(neural_speech_test PRIVATE neural_speech)
add_test(NAME neural_speech COMMAND neural_speech_test)
add_executable(speech_leveler_test native/speech_leveler_test.cpp)
target_link_libraries(speech_leveler_test PRIVATE neural_speech)
add_test(NAME speech_leveler COMMAND speech_leveler_test)
add_executable(audio_render native/audio_render.cpp)
target_link_libraries(audio_render PRIVATE neural_speech speex_preprocess)
